CREATE TABLE DEPARTMENTS (department_id INT GENERATED AS IDENTITY PRIMARY KEY, department_name VARCHAR(100) NOT NULL, manager_id INT, location_id INT);

CREATE TABLE LOCATIONS (location_id INT GENERATED AS IDENTITY PRIMARY KEY, street_address VARCHAR(100), postal_code VARCHAR(10), city VARCHAR(50), state_province VARCHAR(50), country_id INT);

CREATE TABLE JOB_HISTORY (employee_id INT GENERATED AS IDENTITY, start_date DATE, end_date DATE, job_id INT, department_id INT, PRIMARY KEY (employee_id, start_date));

CREATE TABLE EMPLOYEES (employee_id INT GENERATED AS IDENTITY PRIMARY KEY, first_name VARCHAR(50), last_name VARCHAR(50), email VARCHAR(50), phone_number NUMBER(8), hire_date DATE, job_id INT, salary FLOAT, comission_pct FLOAT, manager_id INT, department_id INT);

CREATE TABLE JOBS (job_id INT GENERATED AS IDENTITY PRIMARY KEY, job_title VARCHAR(100), min_salary FLOAT, max_salary FLOAT, CHECK (max_salary - min_salary >= 2000));

CREATE TABLE COUNTIRES (country_id INT GENERATED AS IDENTITY PRIMARY KEY, country_name VARCHAR(100), region_id INT);

CREATE TABLE REGIONS (region_id INT GENERATED AS IDENTITY PRIMARY KEY, region_name VARCHAR(100));
