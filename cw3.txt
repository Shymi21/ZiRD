SELECT 
    LAST_NAME, 
    SALARY,
    RANK() OVER (ORDER BY SALARY DESC) AS ranking_pensji,
    SUM(SALARY) OVER () AS suma_wszystkich_pensji
FROM EMPLOYEES;

CREATE TABLE SALES AS SELECT * FROM HR.SALES;
CREATE TABLE PRODUCTS AS SELECT * FROM HR.PRODUCTS;

SELECT 
    e.LAST_NAME, 
    p.PRODUCT_NAME,
    SUM(s.QUANTITY * s.PRICE) OVER (PARTITION BY e.EMPLOYEE_ID ORDER BY s.SALE_DATE) AS skumulowana_sprzedaz,
    RANK() OVER (ORDER BY s.QUANTITY * s.PRICE DESC) AS ranking_wartosci_zamowienia
FROM EMPLOYEES e
JOIN SALES s ON e.EMPLOYEE_ID = s.EMPLOYEE_ID
JOIN PRODUCTS p ON s.PRODUCT_ID = p.PRODUCT_ID;

SELECT 
    p.PRODUCT_NAME, 
    s.PRICE,
    SUM(s.QUANTITY * s.PRICE) OVER (PARTITION BY s.PRODUCT_ID, TRUNC(s.SALE_DATE, 'MM')) AS suma_miesieczna_calkowita,
    SUM(s.QUANTITY * s.PRICE) OVER (PARTITION BY s.PRODUCT_ID, TRUNC(s.SALE_DATE, 'MM') ORDER BY s.SALE_DATE) AS suma_miesieczna_rosnaca
FROM SALES s
JOIN PRODUCTS p ON s.PRODUCT_ID = p.PRODUCT_ID;

SELECT 
    p.PRODUCT_NAME, 
    p.PRODUCT_CATEGORY,
    s22.SALE_DATE AS data,
    s22.PRICE AS cena_2022, 
    s23.PRICE AS cena_2023,
    (s23.PRICE - s22.PRICE) AS roznica
FROM SALES s22
JOIN SALES s23 ON s22.PRODUCT_ID = s23.PRODUCT_ID 
              AND TO_CHAR(s22.SALE_DATE, 'MM-DD') = TO_CHAR(s23.SALE_DATE, 'MM-DD')
JOIN PRODUCTS p ON s22.PRODUCT_ID = p.PRODUCT_ID
WHERE EXTRACT(YEAR FROM s22.SALE_DATE) = 2022 
  AND EXTRACT(YEAR FROM s23.SALE_DATE) = 2023;

SELECT 
    PRODUCT_CATEGORY, 
    PRODUCT_NAME, 
    PRODUCT_ID, 
    MIN(PRODUCT_ID) OVER (PARTITION BY PRODUCT_CATEGORY) AS min_w_kat,
    MAX(PRODUCT_ID) OVER (PARTITION BY PRODUCT_CATEGORY) AS max_w_kat,
    (MAX(PRODUCT_ID) OVER (PARTITION BY PRODUCT_CATEGORY) - MIN(PRODUCT_ID) OVER (PARTITION BY PRODUCT_CATEGORY)) AS roznica_max_min
FROM PRODUCTS;

SELECT 
    p.PRODUCT_NAME, 
    s.SALE_DATE,
    AVG(s.PRICE) OVER (PARTITION BY s.PRODUCT_ID ORDER BY s.SALE_DATE 
                       ROWS BETWEEN 1 PRECEDING AND 1 FOLLOWING) AS srednia_kroczaca
FROM SALES s
JOIN PRODUCTS p ON s.PRODUCT_ID = p.PRODUCT_ID;

SELECT 
    PRODUCT_NAME, 
    PRODUCT_CATEGORY, 
    PRODUCT_ID AS CENA,
    RANK() OVER (PARTITION BY PRODUCT_CATEGORY ORDER BY PRODUCT_ID) AS ranking,
    ROW_NUMBER() OVER (PARTITION BY PRODUCT_CATEGORY ORDER BY PRODUCT_ID) AS numer_wiersza,
    DENSE_RANK() OVER (PARTITION BY PRODUCT_CATEGORY ORDER BY PRODUCT_ID) AS ranking_gesty
FROM PRODUCTS;

SELECT 
    e.LAST_NAME, 
    p.PRODUCT_NAME,
    (s.PRICE * s.QUANTITY) AS wartosc_wiersza,
    SUM(s.PRICE * s.QUANTITY) OVER (PARTITION BY e.EMPLOYEE_ID ORDER BY s.SALE_DATE) AS wartosc_rosnaca_pracownika,
    RANK() OVER (ORDER BY (s.PRICE * s.QUANTITY) DESC) AS ranking_globalny
FROM SALES s
JOIN EMPLOYEES e ON s.EMPLOYEE_ID = e.EMPLOYEE_ID
JOIN PRODUCTS p ON s.PRODUCT_ID = p.PRODUCT_ID;

SELECT DISTINCT e.FIRST_NAME, e.LAST_NAME, e.JOB_ID
FROM EMPLOYEES e
JOIN SALES s ON e.EMPLOYEE_ID = s.EMPLOYEE_ID;
